// 1. Quản lý người dùng (User Management)
Table users {
  id int [pk, increment]
  username varchar(50) [not null, unique]
  password varchar(255) [not null]
  email varchar(255) [not null, unique]
  full_name varchar(255)
  phone varchar(20)
  address text
  role varchar(20) [default: 'user'] // user, admin, staff
  is_active tinyint [default: 1]
  created_at timestamp [default: `CURRENT_TIMESTAMP`]
  updated_at timestamp [default: `CURRENT_TIMESTAMP`]
  remember_token varchar(64)
  remember_expires datetime
  position varchar(50)
}

// 2. Quản lý sản phẩm (Catalog Management)
Table categories {
  id int [pk, increment]
  name varchar(100) [not null]
  slug varchar(100) [not null, unique]
  description text
  created_at timestamp [default: `CURRENT_TIMESTAMP`]
}

Table products {
  id int [pk, increment]
  name varchar(255) [not null]
  slug varchar(255) [unique]
  price decimal(12,2) [not null]
  old_price decimal(12,2)
  category_id int [not null]
  image varchar(255) [default: 'models_default_img.jpeg']
  description text
  grade varchar(20) [default: 'HG'] // Thay cho 'type' để đúng thuật ngữ Gundam (HG, RG, MG, PG, EG, SD)
  series varchar(100) // Ví dụ: Universal Century, Seed, OO...
  scale varchar(20)   // Ví dụ: 1/144, 1/100, 1/60
  stock int [default: 50]
  is_featured tinyint [default: 0]
  is_sale tinyint [default: 0]
  status varchar(20) [default: 'active'] // active, inactive, out_of_stock
  created_at timestamp [default: `CURRENT_TIMESTAMP`]
  updated_at timestamp [default: `CURRENT_TIMESTAMP`]

  Indexes {
    category_id
    status
    grade
  }
}

// 3. Giỏ hàng & Đơn hàng (Cart & Order Management)
Table cart {
  id int [pk, increment]
  user_id int [not null]
  product_id int [not null]
  quantity int [default: 1]
  added_at timestamp [default: `CURRENT_TIMESTAMP`]
  selected tinyint [not null, default: 1]

  Indexes {
    (user_id, product_id) [unique] // Ngăn trùng lặp sản phẩm trong giỏ của một user
  }
}

Table coupons {
  id int [pk, increment]
  code varchar(50) [not null, unique]
  description varchar(255)
  discount_type varchar(20) [default: 'percent'] // percent, fixed
  discount_value decimal(10,2) [default: 0.00]
  min_order decimal(12,2) [default: 0.00]
  max_uses int
  used_count int [default: 0]
  starts_at datetime
  expires_at datetime
  is_active tinyint [default: 1]
  created_at timestamp [default: `CURRENT_TIMESTAMP`]
}

Table orders {
  id int [pk, increment]
  order_code varchar(20) [not null, unique]
  user_id int [not null]
  full_name varchar(255) [not null]
  phone varchar(20) [not null]
  email varchar(255) [not null]
  address text [not null]
  note text
  subtotal decimal(12,2) [default: 0.00]
  shipping_fee decimal(12,2) [default: 0.00]
  total decimal(12,2) [default: 0.00]
  payment_method varchar(20) [default: 'cod'] // cod, vnpay, momo
  status varchar(20) [default: 'pending']      // pending, processing, shipped, completed, cancelled
  created_at timestamp [default: `CURRENT_TIMESTAMP`]
  updated_at timestamp [default: `CURRENT_TIMESTAMP`]
  coupon_code varchar(50)
  discount_amount decimal(12,2) [default: 0.00]

  Indexes {
    user_id
    order_code
    status
  }
}

Table order_items {
  id int [pk, increment]
  order_id int [not null]
  product_id int [not null]
  product_name varchar(255) [not null] // Lưu snapshot tên sản phẩm đề phòng tương lai đổi tên sản phẩm gốc
  product_image varchar(255)            // Lưu snapshot hình ảnh sản phẩm
  price decimal(12,2) [not null]        // Lưu giá tại thời điểm mua (historical price data)
  quantity int [default: 1]
  subtotal decimal(12,2) [not null]
}

Table order_returns {
  id int [pk, increment]
  order_id int [not null]
  reason text [not null]
  status varchar(20) [default: 'pending'] // pending, approved, rejected
  admin_comment text
  created_at timestamp [default: `CURRENT_TIMESTAMP`]
  updated_at timestamp [default: `CURRENT_TIMESTAMP`]
}

// 4. Hệ thống Chatbot AI (AI Assistant)
Table chat_sessions {
  id int [pk, increment]
  user_id int [not null]
  title varchar(255)
  created_at timestamp [default: `CURRENT_TIMESTAMP`]
  updated_at timestamp [default: `CURRENT_TIMESTAMP`]
}

Table chat_messages {
  id int [pk, increment]
  session_id int [not null]
  user_id int [not null]
  role varchar(20) [not null] // system, user, assistant
  message text [not null]
  created_at timestamp [default: `CURRENT_TIMESTAMP`]
}

// 5. Đánh giá & Tương tác (Feedback System)
Table reviews {
  id int [pk, increment]
  product_id int [not null]
  user_id int [not null]
  rating tinyint [not null] // Constraint từ 1 -> 5
  comment text
  created_at timestamp [default: `CURRENT_TIMESTAMP`]
}

// 6. Thông báo (Notification System)
// Phân tách rõ ràng: notifications chứa nội dung. notification_users lưu trạng thái nhận/đọc của từng người.
Table notifications {
  id int [pk, increment]
  title varchar(255) [not null]
  message text [not null]
  type varchar(50) [default: 'system'] // system, personal, order_update
  created_at timestamp [default: `CURRENT_TIMESTAMP`]
}

Table notification_users {
  id int [pk, increment]
  user_id int [not null]
  notification_id int [not null]
  is_read tinyint [default: 0]
  read_at timestamp

  Indexes {
    (user_id, notification_id) [unique]
  }
}
